export type AppLocale = 'AU' | 'UK';

export async function detectLocale(): Promise<AppLocale> {
  // Signal 1: Device/browser language locale (VPN-proof)
  const deviceLocale =
    (typeof navigator !== 'undefined' && navigator.language) || '';

  if (deviceLocale.toLowerCase().includes('en-gb')) return 'UK';
  if (deviceLocale.toLowerCase().includes('en-au')) return 'AU';

  // Signal 2: IP geolocation fallback (catches remaining cases)
  try {
    const res = await fetch('https://ipapi.co/json/', { signal: AbortSignal.timeout(3000) });
    const data = await res.json();
    if (data?.country_code === 'GB') return 'UK';
    if (data?.country_code === 'AU') return 'AU';
  } catch {
    // Network or timeout — fall through to default
  }

  // Default: AU (primary market)
  return 'AU';
}

export function getStoredLocale(): AppLocale | null {
  try {
    const stored = localStorage.getItem('sortii_locale');
    if (stored === 'AU' || stored === 'UK') return stored;
  } catch {}
  return null;
}

export function storeLocale(locale: AppLocale): void {
  try {
    localStorage.setItem('sortii_locale', locale);
  } catch {}
}
```

import { AppLocale } from './locale';
import auRecipes from '../data/recipes_unified_au.json';
import ukRecipes from '../data/recipes_uk_clean.json';

export function getRecipesForLocale(locale: AppLocale) {
  if (locale === 'UK') {
    // Filter AU file: only recipes where locale includes 'UK'
    const universalRecipes = (auRecipes as any[])
      .filter(r => Array.isArray(r.locale) && r.locale.includes('UK'))
      .map(r => ({
        ...r,
        // Serve UK ingredient names
        ingredients: r.ingredientsUK ?? r.ingredients,
        // Prefer GBP pricing field
        costPerServe: r.costPerServeGBP ?? null,
        currency: 'GBP',
        currencySymbol: '£',
      }));

    // Add UK-specific recipes
    const ukOnlyRecipes = (ukRecipes as any[]).map(r => ({
      id: r.recipeId,
      name: r.title,
      locale: ['UK'],
      ingredients: r.ingredientRows ?? [],
      costPerServe: r.estimatedCostPerServeGBP ?? null,
      currency: 'GBP',
      currencySymbol: '£',
      cookTimeMinutes: r.cookTimeMins,
      tags: r.tags ?? [],
      mealType: r.mealType ?? 'dinner',
      source: 'uk_pack',
      pricingNote: r.pricingNote ?? null,
      // Pass through all other fields
      ...r,
    }));

    return [...universalRecipes, ...ukOnlyRecipes];
  }

  // AU: return all recipes with AU ingredients and AUD pricing
  return (auRecipes as any[]).map(r => ({
    ...r,
    ingredients: r.ingredientsAU ?? r.ingredients,
    costPerServe: r.costPerServeAUD ?? null,
    currency: 'AUD',
    currencySymbol: 'A$',
  }));
}

export function getCostDisplay(recipe: any, locale: AppLocale): string {
  const symbol = locale === 'UK' ? '£' : 'A$';
  const cost = recipe.costPerServe;
  if (!cost) return '';
  return `${symbol}${cost.toFixed(2)}/serve`;
}

export function getWeeklyCost(
  plannedMeals: any[],
  householdSize: number,
  locale: AppLocale
): { low: number; high: number; total: number } {
  const total = plannedMeals.reduce((sum, meal) => {
    const cost = meal.costPerServe ?? fallbackCost(meal.estimatedCostBand, locale);
    return sum + cost * householdSize;
  }, 0);

  return {
    low: Math.round(total * 0.92),
    high: Math.round(total * 1.08),
    total: Math.round(total),
  };
}

function fallbackCost(band: string, locale: AppLocale): number {
  // Fallback cost per serve when recipe has no pricing
  if (locale === 'UK') {
    if (band === 'low') return 1.20;
    if (band === 'high') return 3.00;
    return 1.80;
  }
  // AU fallbacks
  if (band === 'low') return 4.50;
  if (band === 'high') return 11.00;
  return 7.00;
}
```

async function priceBatch(recipes) {
  const recipeData = recipes.map(r => ({
    recipeId: r.recipeId,
    title: r.title,
    serves: r.serves || 4,
    ingredients: r.ingredientRows || [],
  }));

  const userMessage = `Price these UK recipes. Return only the JSON array.

${JSON.stringify(recipeData, null, 2)}`;

  const response = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 1000,
    system: SYSTEM_PROMPT,
    messages: [{ role: 'user', content: userMessage }],
  });

  const raw = response.content[0]?.text || '';

  // Strip any accidental markdown fences
  const cleaned = raw.replace(/```json|```/g, '').trim();

  return JSON.parse(cleaned);
}
