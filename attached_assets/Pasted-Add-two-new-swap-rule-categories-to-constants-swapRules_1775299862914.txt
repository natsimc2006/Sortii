─── PART 2: Frequency-based tip for the weekly plan ───

In utils/getSwapsForRecipe.ts, add a separate exported function:

export function getWeeklyPantryTips(allPlannedMeals: Recipe[]): string[] {

  const tips: string[] = [];
  const allIngredients = allPlannedMeals
    .flatMap(m => m.ingredients.map(i => i.name.toLowerCase()));

  // Count garlic occurrences across the week
  const garlicCount = allIngredients
    .filter(i => i.includes('garlic')).length;
  if (garlicCount >= 3) {
    tips.push(
      `Garlic features in ${garlicCount} of your meals this week. ` +
      `A jar of minced garlic (~$3) or garlic powder covers the whole week ` +
      `and keeps for months.`
    );
  }

  // Count fresh herb occurrences
  const freshHerbCount = allIngredients
    .filter(i => 
      i.includes('fresh') || 
      ['basil','coriander','parsley','thyme','rosemary','dill']
        .some(h => i.includes(h))
    ).length;
  if (freshHerbCount >= 3) {
    tips.push(
      `You have ${freshHerbCount} fresh herb requirements this week. ` +
      `Dried herbs at ⅓ quantity work just as well in cooked dishes ` +
      `and cost a fraction of fresh bunches.`
    );
  }

  // Count onion occurrences
  const onionCount = allIngredients
    .filter(i => i.includes('onion')).length;
  if (onionCount >= 4) {
    tips.push(
      `Onion appears in ${onionCount} meals. Buy a 1kg bag (~$2.50) ` +
      `rather than individual onions — much better value.`
    );
  }

  return tips;
}
