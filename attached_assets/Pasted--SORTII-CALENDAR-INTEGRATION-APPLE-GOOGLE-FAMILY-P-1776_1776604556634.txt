── CREATE: /utils/appleCalendar.ts ───────

import * as Calendar from 'expo-calendar';

export async function requestCalendarPermission(): 
  Promise<boolean> {
  const { status } = await 
    Calendar.requestCalendarPermissionsAsync();
  return status === 'granted';
}

export async function getAppleCalendars(): 
  Promise<Calendar.Calendar[]> {
  const calendars = await 
    Calendar.getCalendarsAsync(
      Calendar.EntityTypes.EVENT
    );
  return calendars;
}

export async function getAppleEventsForRange(
  startDate: Date,
  endDate: Date,
  calendarIds?: string[]
): Promise<SortiiEvent[]> {
  
  // If no specific calendars, get all
  let ids = calendarIds;
  if (!ids || ids.length === 0) {
    const cals = await getAppleCalendars();
    ids = cals.map(c => c.id);
  }

  const events = await 
    Calendar.getEventsAsync(
      ids,
      startDate,
      endDate
    );

  // Map to Sortii event format
  return events.map(e => ({
    id: `apple_${e.id}`,
    source: 'apple',
    title: e.title || 'Untitled event',
    startTime: e.startDate,
    endTime: e.endDate,
    location: e.location || null,
    notes: e.notes || null,
    calendarName: e.calendar?.title || null,
    calendarColor: e.calendar?.color || null,
    isAllDay: e.allDay || false,
    pushToFamily: false,
    addedToFamilyJuggle: false,
  }));
}

// Get events for current week + next 2 weeks
export async function getUpcomingAppleEvents(
  calendarIds?: string[]
): Promise<SortiiEvent[]> {
  const now = new Date();
  const twoWeeksOut = new Date();
  twoWeeksOut.setDate(now.getDate() + 14);
  
  return getAppleEventsForRange(
    now,
    twoWeeksOut,
    calendarIds
  );
}

── CREATE: /utils/googleCalendar.ts ──────

import * as AuthSession from 'expo-auth-session';
import * as WebBrowser from 'expo-web-browser';

WebBrowser.maybeCompleteAuthSession();

const GOOGLE_CLIENT_ID = 
  process.env.GOOGLE_CLIENT_ID || 
  'GOOGLE_CLIENT_ID_NOT_SET';

const SCOPES = [
  'https://www.googleapis.com/auth/calendar.readonly',
  'https://www.googleapis.com/auth/userinfo.email',
];

export async function signInWithGoogle(): 
  Promise<string | null> {
  
  const discovery = {
    authorizationEndpoint: 
      'https://accounts.google.com/o/oauth2/v2/auth',
    tokenEndpoint: 
      'https://oauth2.googleapis.com/token',
  };

  const redirectUri = 
    AuthSession.makeRedirectUri({ 
      useProxy: true 
    });

  const request = new AuthSession.AuthRequest({
    clientId: GOOGLE_CLIENT_ID,
    scopes: SCOPES,
    redirectUri,
    responseType: 
      AuthSession.ResponseType.Token,
  });

  const result = await request.promptAsync(
    discovery
  );

  if (result.type === 'success') {
    const token = result.params.access_token;
    // Store token
    await AsyncStorage.setItem(
      'sortii_google_token', 
      token
    );
    return token;
  }
  
  return null;
}

export async function getGoogleCalendars(
  token: string
): Promise<GoogleCalendar[]> {
  const response = await fetch(
    'https://www.googleapis.com/calendar/v3/users/me/calendarList',
    {
      headers: { 
        Authorization: `Bearer ${token}` 
      }
    }
  );
  
  if (!response.ok) {
    throw new Error('Google Calendar auth failed');
  }
  
  const data = await response.json();
  return data.items || [];
}

export async function getGoogleEventsForRange(
  token: string,
  startDate: Date,
  endDate: Date,
  calendarIds: string[]
): Promise<SortiiEvent[]> {
  
  const allEvents: SortiiEvent[] = [];
  
  for (const calendarId of calendarIds) {
    const params = new URLSearchParams({
      timeMin: startDate.toISOString(),
      timeMax: endDate.toISOString(),
      singleEvents: 'true',
      orderBy: 'startTime',
      maxResults: '50',
    });

    const response = await fetch(
      `https://www.googleapis.com/calendar/v3/calendars/${
        encodeURIComponent(calendarId)
      }/events?${params}`,
      {
        headers: { 
          Authorization: `Bearer ${token}` 
        }
      }
    );

    if (!response.ok) continue;
    
    const data = await response.json();
    
    const events = (data.items || []).map(
      (e: any) => ({
        id: `google_${e.id}`,
        source: 'google' as const,
        title: e.summary || 'Untitled',
        startTime: 
          e.start?.dateTime || 
          e.start?.date || '',
        endTime: 
          e.end?.dateTime || 
          e.end?.date || null,
        location: e.location || null,
        notes: e.description || null,
        calendarName: calendarId,
        calendarColor: null,
        isAllDay: !!e.start?.date && 
          !e.start?.dateTime,
        pushToFamily: false,
        addedToFamilyJuggle: false,
      })
    );
    
    allEvents.push(...events);
  }
  
  return allEvents.sort((a, b) => 
    new Date(a.startTime).getTime() - 
    new Date(b.startTime).getTime()
  );
}

export async function getUpcomingGoogleEvents(
  token: string,
  calendarIds: string[]
): Promise<SortiiEvent[]> {
  const now = new Date();
  const twoWeeksOut = new Date();
  twoWeeksOut.setDate(now.getDate() + 14);
  return getGoogleEventsForRange(
    token, now, twoWeeksOut, calendarIds
  );
}

── TOKEN REFRESH HANDLING ────────────────

Google tokens expire after 1 hour.
Add this check before any Google API call:

async function getValidGoogleToken(): 
  Promise<string | null> {
  const token = await AsyncStorage.getItem(
    'sortii_google_token'
  );
  if (!token) return null;
  
  // Test if token is still valid
  const test = await fetch(
    'https://www.googleapis.com/oauth2/v1/tokeninfo' +
    `?access_token=${token}`
  );
  
  if (test.ok) return token;
  
  // Token expired — prompt re-auth
  return null;
}

If token is null or expired:
Show a quiet banner in the Appointments tab:
"Google Calendar needs reconnecting."
[Reconnect →]

── CREATE: /utils/appointmentService.ts ──

Merges events from all three sources
into a single sorted list.

export async function getAllUpcomingEvents(
  settings: CalendarSettings
): Promise<SortiiEvent[]> {
  
  const allEvents: SortiiEvent[] = [];
  
  // Source 1: Apple calendar
  if (settings.appleConnected && 
      settings.appleCalendarIds.length > 0) {
    try {
      const appleEvents = 
        await getUpcomingAppleEvents(
          settings.appleCalendarIds
        );
      allEvents.push(...appleEvents);
    } catch (e) {
      console.warn('Apple calendar read failed', e);
    }
  }
  
  // Source 2: Google Calendar
  if (settings.googleConnected && 
      settings.googleToken) {
    try {
      const token = 
        await getValidGoogleToken();
      if (token) {
        const googleEvents = 
          await getUpcomingGoogleEvents(
            token,
            settings.googleCalendarIds
          );
        allEvents.push(...googleEvents);
      }
    } catch (e) {
      console.warn('Google calendar read failed', e);
    }
  }
  
  // Source 3: Manual events from AsyncStorage
  const manualRaw = await AsyncStorage.getItem(
    'sortii_manual_events'
  );
  if (manualRaw) {
    const manualEvents = JSON.parse(manualRaw);
    allEvents.push(...manualEvents);
  }
  
  // Deduplicate (same title + same time)
  const seen = new Set<string>();
  const deduped = allEvents.filter(e => {
    const key = `${e.title}_${e.startTime}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
  
  // Sort by start time
  return deduped.sort((a, b) => 
    new Date(a.startTime).getTime() - 
    new Date(b.startTime).getTime()
  );
}
