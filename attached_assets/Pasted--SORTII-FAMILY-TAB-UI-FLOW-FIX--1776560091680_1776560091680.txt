STEP 4 — FIX THE HANDLER FUNCTION
════════════════════════════════════════════

Confirm handleSortJuggle exists and
contains this complete logic.
Replace it entirely with this if needed:

const handleSortJuggle = async () => {
  setIsJuggleSolving(true);
  setShowJuggleResults(false);
  
  // Rotating processing messages
  const messages = [
    'Untangling the family juggle…',
    'Working out who goes where.',
    'Finding the least painful version.',
    'Checking if anyone gets a breather.',
  ];
  let msgIndex = 0;
  const msgInterval = setInterval(() => {
    msgIndex = (msgIndex + 1) % messages.length;
    setProcessingMessage(messages[msgIndex]);
  }, 2500);

  try {
    // Build the prompt from current data
    const adults = familyMembers.filter(m => 
      m.role === 'adult' || 
      m.role === 'Adult/driver' ||
      m.role === 'Adult'
    );
    const children = familyMembers.filter(m => 
      m.role === 'child' || 
      m.role === 'Child'
    );

    // Format adults string
    const adultsStr = adults.map(a => 
      `${a.name}, can drive: yes`
    ).join('\n');

    // Format events string
    const eventsStr = events.map(e => `
Child: ${e.personName || e.person || 'unspecified'}
Activity: ${e.title || e.activity || 'event'}
Location: ${e.location || 'not specified'}
Day: ${e.day || 'Saturday'}
Start: ${e.startTime || e.time || 'not specified'}
End: ${e.endTime || 'not specified'}
Assigned adult: unassigned
Notes: none`).join('\n');

    const prompt = `Available adults:\n${adultsStr}
    
Weekend events:\n${eventsStr}

Please solve this schedule.`;

    // Call the juggle API
    const response = await fetch('/api/juggle', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message: prompt }),
    });

    if (!response.ok) {
      throw new Error(`API error: ${response.status}`);
    }

    const data = await response.json();
    
    // Handle both wrapped and unwrapped responses
    const result = data.result || data;
    
    setJuggleResult(result);
    setShowJuggleResults(true);
    
    // Save to AsyncStorage
    await AsyncStorage.setItem(
      'sortii_last_juggle_result',
      JSON.stringify(result)
    );

  } catch (error) {
    console.error('Juggle solve error:', error);
    Alert.alert(
      'That one got complicated.',
      'Too many moving parts at once. Try again — it usually works second time.',
      [
        { text: 'Try again', onPress: handleSortJuggle },
        { text: 'Not now', style: 'cancel' },
      ]
    );
  } finally {
    clearInterval(msgInterval);
    setIsJuggleSolving(false);
    setProcessingMessage('Untangling the family juggle…');
  }
};
